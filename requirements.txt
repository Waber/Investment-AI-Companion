# Runtime dependencies for the API.
#
# These are direct pins only. Packages the application does not import
# (pandas, numpy, beautifulsoup4, and the rest of the yfinance stack)
# are not pinned here. yfinance installs them. The lock file records
# the exact versions CI and local installs use.
#
# Python 3.12. Regenerate the hashed lock from the repository root:
#
#   python -m piptools compile --generate-hashes --strip-extras --output-file=requirements.lock requirements.txt
#
# --strip-extras does not change this runtime set's pins or hashes.
# requirements.lock was left as compiled without that flag. The next
# regeneration should pass it anyway.
#
# Install the lock, not this file, when you need the same set CI uses:
#
#   python -m pip install --require-hashes -r requirements.lock

# API framework.
# fastapi 0.104.1 pulled starlette 0.27.0 (CVE-2024-24762,
# CVE-2024-47874, CVE-2025-54121). 0.135.2 is the first release line
# whose own floor is starlette>=0.46.0 and pydantic>=2.9.0,<3.
# The starlette floor below is higher than FastAPI's, so resolution
# cannot stop on 0.46.x or 0.47.1.
fastapi>=0.135.2
# CVE-2024-24762 is fixed in 0.36.2, CVE-2024-47874 in 0.40.0,
# CVE-2025-54121 in 0.47.2. 1.3.1 is the floor that also clears the
# later Starlette advisories pip-audit reports for the 0.x line
# (through CVE-2026-54283).
starlette>=1.3.1
uvicorn==0.24.0
python-dotenv>=1.2.2
# Stay on Pydantic v2. FastAPI 0.135.2+ requires >=2.9.0.
pydantic>=2.9.0,<3
pydantic-settings==2.1.0
# 2.31.0 is the old pin. yfinance imports requests. The floor is the
# security minimum from the review.
requests>=2.33.0

# Database. The application imports SQLAlchemy and Alembic.
# psycopg2 is the PostgreSQL driver SQLAlchemy loads by URL; the
# application does not import the module by name.
sqlalchemy==2.0.23
psycopg2-binary==2.9.9
alembic==1.12.1

# Market data. The collector imports yfinance. Tests fake Ticker and
# do not call Yahoo.
yfinance>=1.7
